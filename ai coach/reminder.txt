def build_prompt(intent, filtered_exercise_library):

    # 规范了几类训练目标和强度偏好（休息时长、训练组数），分别对应不同的训练参数范围
    goal_policy = {
 "fat_loss": {
  "rest_range": [10,30],
  "sets_range": [3,5]
  
 },

 "strength": {
  "rest_range": [90,120],
  "sets_range": [3,5]
 },

 "general_fitness": {
  "rest_range": [30,60],
  "sets_range": [3,4]
 }
}
    # 如果有特殊要求（如高强度训练），可以通过调整训练参数来满足用户的偏好
    intensity_modifier = {
 "low": {
  "sets_delta": -1,
  "rest_multiplier": 1.3
 },

 "moderate": {
  "sets_delta": 0,
  "rest_multiplier": 1.0
 },

 "high": {
  "sets_delta": 1,
  "rest_multiplier": 0.8
 }
}

    # 根据用户的训练目标和强度偏好，计算出适合的训练参数范围
    policy = goal_policy[intent["session_goal"]]
    modifier = intensity_modifier[intent["intensity_preference"]]

    sets_range = [
    policy["sets_range"][0] + modifier["sets_delta"],
    policy["sets_range"][1] + modifier["sets_delta"]
    ]
    rest_range = [ 
    int(policy["rest_range"][0] * modifier["rest_multiplier"]),
    int(policy["rest_range"][1] * modifier["rest_multiplier"])
    ]

    max_exercises = intent["duration_minutes"] // 4

    prompt = f"""
You are an AI fitness coach.

Your task is to generate a safe and effective workout plan.

User training intent is:
{json.dumps(intent, indent=2)}

Recommend intensity:
rest_range: {rest_range[0]}-{rest_range[1]} seconds
sets_range: {sets_range[0]}-{sets_range[1]} sets

Available exercise library:
{json.dumps(filtered_exercise_library, indent=2)}

Use about {max_exercises} exercises.

Requirements:
1. Only use exercises from the exercise library
2. Match the requested workout duration
3. Follow the preferred intensity
4. Respect physical constraints (injuries or avoid_muscles)
5. The workout should be suitable for general fitness users

Output format (JSON only):

{{
 "workout_plan":[
  {{
   "exercise": "exercise_name",
   "total_sets": 3
   "rest_seconds": 30
   "avg_set_time": 40
  }},
  {{
   "exercise": "exercise_name",
   "total_sets": 4
   "rest_seconds": 20
   "avg_set_time": 20
  }}
 ]
}}

Do not output anything except JSON.
"""

    return prompt
